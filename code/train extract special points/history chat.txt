pythondef extract_bod_records(do_series):
    # do_series: pd.Series, index = second (1,2,3,...), value = DO (mV)
    
    start_second = 301
    interval = 473
    window_min = 60
    window_max = 120
    
    peaks = []
    t = start_second
    while t + window_max < len(do_series):
        do_in = do_series.iloc[t-1]  # vì index bắt đầu từ 1
        do_min = do_series.iloc[t + window_min - 1 : t + window_max].min()
        ddo = do_in - do_min
        
        peaks.append({
            'No.peak': t,
            'Doin': do_in,
            'DOmin': do_min,
            'DDO': ddo
        })
        
        t += interval
    
    return pd.DataFrame(peaks)

def extract_bod_records(do_series, sample_name):
    candidates = []
    for start in range(290, 321):           # 301 ± 15
        for interval in [472, 473, 474, 475]: # ±2s quanh 473
            t = start
            records = []
            while t + 130 < len(do_series):
                try:
                    do_in = do_series.iloc[t - 1]
                    segment = do_series.iloc[t + 60 : t + 130]
                    if len(segment) == 0:

                    do_min = segment.min()

                        'Sample Name': sample_name

                except:
                    break
                t += interval
            
            if 10 <= len(records) <= 25:  # số record hợp lý

                candidates.append((score, start, interval, records))

        return None
    
    best = max(candidates, key=lambda x: x[0])

    mid = len(records) // 2
    for i, r in enumerate(records):

# ================== SMART NAME MAPPER ==================
from difflib import SequenceMatcher
import re

def normalize_name(s):
    s = s.lower()
    s = re.sub(r'[^\w\s\-\.=]', ' ', s)
    s = re.sub(r'\s+', ' ', s)
    s = re.sub(r'_', '/', s)
    s = re.sub(r'phut', 'phút', s)
    s = re.sub(r'ml', 'ml', s)
    s = re.sub(r'q=', 'q=', s)
    return s.strip()

def smart_match_name(txt_name, excel_candidates):
    txt_norm = normalize_name(txt_name)
    
    best_score = 0
    best_match = None
    
    for cand in excel_candidates:
        cand_norm = normalize_name(cand)
        
        # 1. Token set ratio (rất mạnh với thứ tự khác)
        ratio1 = SequenceMatcher(None, txt_norm, cand_norm).ratio()
        
        # 2. Partial ratio (Q=49.81 khớp Q=49.66)
        ratio2 = max(
            SequenceMatcher(None, txt_norm, cand_norm[i:i+len(txt_norm)]).ratio()
            for i in range(len(cand_norm) - len(txt_norm) + 1)
        ) if len(cand_norm) > len(txt_norm) else ratio1
        
        # 3. Tìm số chung: BOD15, BOD10, Q=50.02, ngày tháng
        numbers_txt = set(re.findall(r'\d+\.?\d*', txt_norm))
        numbers_cand = set(re.findall(r'\d+\.?\d*', cand_norm))
        num_match = len(numbers_txt & numbers_cand) / max(len(numbers_txt), 1)
        
        # 4. Từ khóa: BOD, Zn, Ni, Cr, VS2
        keywords = ['bod', 'zn', 'ni', 'cr', 'vs2', 'vs1']
        kw_score = sum(1 for kw in keywords if kw in txt_norm and kw in cand_norm) / 3
        
        # Tổng score
        score = (ratio1 * 0.4 + ratio2 * 0.3 + num_match * 0.2 + kw_score * 0.1)
        
        if score > best_score:
            best_score = score
            best_match = cand
    
    return best_match, best_score

def smart_match_name(txt_name, candidates):
    txt_norm = normalize_name(txt_name)
    best_score = 0
    best_match = None
    for cand in candidates:
        cand_norm = normalize_name(cand)
        ratio = SequenceMatcher(None, txt_norm, cand_norm).ratio()
        partial = max(SequenceMatcher(None, txt_norm, cand_norm[i:i+len(txt_norm)]).ratio()

def extract_bod(do_series):
    print(f"   Dữ liệu có {len(do_series)} điểm → ", end="")
    if len(do_series) < 7000:
        print("QUÁ NGẮN! File bị cắt → thử với start sớm hơn...")
    
    best_records = None
    best_score = -1
    best_params = None
    
    # Mở rộng phạm vi tìm kiếm vì file có thể bị cắt
    for start in range(280, 350):           # 301 ±70
        for interval in [472, 473, 474, 475, 471]:
            t = start
            records = []
            peaks_found = 0
            while t + 130 < len(do_series) and peaks_found < 20:
                try:
                    do_in = do_series.iloc[t-1]
                    segment = do_series.iloc[t+50:t+140]  # mở rộng cửa sổ
                    if len(segment) < 20:
                        break
                    do_min = segment.min()
                    ddo = do_in - do_min
                    if 0.5 < ddo < 15:  # nới lỏng điều kiện
                        records.append({
                            'No.peak': t,
                            'Doin (mV)': round(do_in, 5),
                            'DOmin (mV)': round(do_min, 5),
                            'DDO (mV)': round(ddo, 5)
                        })
                        peaks_found += 1
                except:
                    break
                t += interval
            
            # Ưu tiên đúng 15 điểm, nhưng vẫn chấp nhận 10-16 nếu gần đúng
            if 10 <= len(records) <= 20:
                score = len(records) * 1000
                if len(records) == 15:
                    score += 10000
                if abs(interval - 473) <= 1:
                    score += 5000
                if 295 <= start <= 310:
                    score += 3000
                
                if score > best_score:
                    best_score = score
                    best_records = records.copy()
                    best_params = (start, interval, len(records))
    
    if best_records is not None:
        print(f"THÀNH CÔNG! Tìm thấy {len(best_records)} điểm → start={best_params[0]}, interval={best_params[1]}")
        df = pd.DataFrame(best_records)
        mid = len(df) // 2
        df['Tag'] = ['BOD5'] * mid + ['BOD0'] * (len(df) - mid)
        return df, best_params
    else:
        print("THẤT BẠI! Không tìm thấy chu kỳ nào hợp lệ.")
        return None, None

# Extract hybrid: rule + peak finding
def extract_bod(do_series):
    if len(do_series) < 9000:
        print("WARNING: TXT bị truncated, chỉ extract được phần đầu.")
    
    # Step 1: Tìm local max as potential peaks
    peaks, _ = find_peaks(do_series, height=do_series.mean() + do_series.std()/2, distance=400)  # min distance 400s
    peaks = peaks + 1  # 1-based
    
    # Step 2: Lọc peaks gần chu kỳ 473s from start ~301
    candidate_starts = peaks[(peaks > 280) & (peaks < 320)]
    if len(candidate_starts) == 0:
        start = 301
    else:
        start = candidate_starts[0]
    
    records = []
    t = start
    while t + 140 < len(do_series):
        do_in = do_series[t]
        segment = do_series[t+50:t+140]
        if len(segment) < 20:
            break
        do_min = segment.min()
        ddo = do_in - do_min
        if 3 < ddo < 6.5:  # based on new truth
            records.append({
                'No.peak': t,
                'Doin (mV)': round(do_in, 5),
                'DOmin (mV)': round(do_min, 5),
                'DDO (mV)': round(ddo, 5)
            })
        # Dynamic interval: find next peak
        next_peaks = peaks[(peaks > t + 470) & (peaks < t + 480)]
        if len(next_peaks) == 0:
            break
        t = next_peaks[0]
    
    if len(records) < 18:
        return None
    
    df = pd.DataFrame(records)
    df['Tag'] = ['BOD10'] * 8 + ['BOD5'] * (len(df) - 8)
    return df

# SIÊU HÀM NORMALIZE + MATCH
def normalize(s):
    if pd.isna(s):
        return ""
    s = str(s).lower()
    # Xóa prefix N4-, U96-, VS2-, ngày tháng
    s = re.sub(r'^[a-zA-Z0-9\-]+-', '', s)
    s = re.sub(r'\d{8}-', '', s)
    s = re.sub(r'\d{6}-', '', s)
    # Chuẩn hóa
    s = re.sub(r'bod ?(\d+)', r'bod\1', s)
    s = re.sub(r'q=? ?', 'q=', s)
    s = re.sub(r'ml/phút', 'ml/phút', s)
    s = re.sub(r'_', '/', s)
    s = re.sub(r'[^\w\.\-/=]', ' ', s)
    s = re.sub(r'\s+', ' ', s)
    s = s.replace('.txt', '').strip()
    return s

def match_score(a, b):
    a_norm = normalize(a)
    b_norm = normalize(b)
    
    # 1. Exact match sau normalize
    if a_norm == b_norm:
        return 1.0, "EXACT"
    
    # 2. Sequence ratio
    ratio = SequenceMatcher(None, a_norm, b_norm).ratio()
    
    # 3. Số chung (BOD10, Q=49.81, 01042024)
    nums_a = set(re.findall(r'bod\d+|q=[\d\.]+|\d{8}', a_norm))
    nums_b = set(re.findall(r'bod\d+|q=[\d\.]+|\d{8}', b_norm))
    num_score = len(nums_a & nums_b) / max(len(nums_a), 1)
    
    # 4. Partial match
    partial = 0
    for part in a_norm.split():
        if part in b_norm:
            partial += len(part)
    partial /= len(a_norm)
    
    score = ratio * 0.5 + num_score * 0.3 + partial * 0.2
    return score, "FUZZY"

# BẮT ĐẦU MATCH

python# -*- coding: utf-8 -*-
"""
PREDICT + EXTRACT - CHẠY 300 FILE TRONG 2 PHÚT
Output Excel đẹp
"""

import pandas as pd
import numpy as np
from xgboost import XGBRegressor
import os

# ĐƯỜNG DẪN
TXT_FOLDER = r"E:\VHL Project\Bio Zone\VHL_Biology\data\txt_files"  # folder 300 TXT
OUTPUT_EXTRACT = r"E:\VHL Project\Bio Zone\VHL_Biology\data\SPECIAL_POINTS_FULL.xlsx"

model = XGBRegressor()
model.load_model('bod_model.json')

results = []

for txt_file in os.list_dir(TXT_FOLDER):
    if not txt_file.endswith('.txt'):
        continue
    df_txt = pd.read_csv(os.path.join(TXT_FOLDER, txt_file), sep='\s+', names=['Time', 'DO'])
    do_series = df_txt['DO'].values[:5000]
    
    # Features same as train
    features = np.concatenate([
        [np.mean(do_series), np.std(do_series), np.ptp(do_series)],
        np.percentile(do_series, [10, 25, 50, 75, 90]),
        [len(do_series)],
        [len(find_peaks(do_series[:1000], distance=50)[0])],
        [np.diff(do_series[:200]).std()]
    ]).reshape(1, -1)
    
    pred = model.predict(features)[0]
    start = int(pred[0])
    interval = pred[1]
    window_offset = int(pred[2])
    
    # Extract
    records = []
    t = start
    while t + window_offset + 70 < len(do_series):  # dynamic window
        do_in = do_series[t - 1]
        segment = do_series[t + window_offset - 1 : t + window_offset + 70]
        do_min = segment.min()
        ddo = do_in - do_min
        records.append({
            'No.peak': t,
            'Doin (mV)': round(do_in, 5),
            'DOmin (mV)': round(do_min, 5),
            'DDO (mV)': round(ddo, 5),
            'Sample Name': txt_file
        })
        t += int(interval)
    
    # Tag auto based on number
    mid = len(records) // 2
    for i, r in enumerate(records):
        r['Tag'] = 'BOD10' if i < mid else 'BOD5'  # adjust based on your pattern
    
    results.append(pd.DataFrame(records))

final = pd.concat(results, ignore_index=True)
final.to_excel(OUTPUT_EXTRACT, index=False)

print(f"HOÀN TẤT 300 FILE! OUTPUT TẠI: {OUTPUT_EXTRACT}")
